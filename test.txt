      const options = { weekday: 'long', year: 'numeric', month: 'long', day: 'numeric' };
      document.getElementById('header-date').innerText = new Date().toLocaleDateString('he-IL', options);
    }

    // Tab view switcher
    function switchTab(tabName) {
      document.querySelectorAll('.app-tab').forEach(el => el.classList.add('hidden'));
      document.getElementById(`tab-${tabName}`).classList.remove('hidden');

      // Update active tab button styles
      const tabs = ['home', 'tasks', 'finance', 'notes'];
      tabs.forEach(t => {
        const btn = document.getElementById(`nav-${t}`);
        if(t === tabName) {
          btn.classList.add('text-blue-400');
          btn.classList.remove('text-slate-400');
          btn.querySelector('span').classList.add('font-bold');
        } else {
          btn.classList.remove('text-blue-400');
          btn.classList.add('text-slate-400');
          btn.querySelector('span').classList.remove('font-bold');
        }
      });
    }

    // Toast alerts wrapper
    function showToast(msg) {
      const toast = document.getElementById('toast');
      toast.innerText = msg;
      toast.classList.remove('opacity-0', 'pointer-events-none', 'translate-y-2');
      toast.classList.add('opacity-100', 'translate-y-0');
      setTimeout(() => {
        toast.classList.add('opacity-0', 'pointer-events-none', 'translate-y-2');
        toast.classList.remove('opacity-100', 'translate-y-0');
      }, 2500);
    }

    // --- COPY CODE ENGINE FOR MOBILE DEVICES ---
    function copyAppCode() {
      const textarea = document.createElement('textarea');
      // Extracting the fully compiled live outerHTML with proper doctype wrapper
      textarea.value = "<" + "!DOCTYPE html>\n" + document.documentElement.outerHTML;
      document.body.appendChild(textarea);
      textarea.select();
      textarea.setSelectionRange(0, 99999); // For maximum iOS/Android browser support
      
      try {
        const success = document.execCommand('copy');
        if(success) {
          showToast("📋 קוד האפליקציה המלא הועתק ללוח המכשיר!");
        } else {
          showToast("העתקה נכשלה במערכת הנוכחית");
        }
      } catch (err) {
        showToast("שגיאה בגישה ללוח המכשיר");
      }
      document.body.removeChild(textarea);
    }

    // --- WATER FEATURE ---
    function addWaterCard() {
      if(state.water >= 12) {
        showToast("הגעת למקסימום המומלץ למעקב!");
        return;
      }
      state.water += 1;
      localStorage.setItem('hub_water', state.water);
      updateWaterUI();
      if(state.water === 8) showToast("🎉 כל הכבוד! הגעת ליעד המים היומי שלך!");
    }

    function updateWaterUI() {
      document.getElementById('water-count').innerText = `${state.water} / 8`;
      const pct = Math.min((state.water / 8) * 100, 100);
      document.getElementById('water-bar').style.width = `${pct}%`;
    }

    // --- TODO FEATURE ---
    function addTodo() {
      const input = document.getElementById('input-todo');
      if(!input.value.trim()) return;

      const newTodo = {
        id: Date.now(),
        text: input.value.trim(),
        done: false
      };
      state.todos.push(newTodo);
      localStorage.setItem('hub_todos', JSON.stringify(state.todos));
      input.value = '';
      renderTodos();
      showToast("המשימה התווספה בהצלחה");
    }

    function toggleTodo(id) {
      state.todos = state.todos.map(t => t.id === id ? {...t, done: !t.done} : t);
      localStorage.setItem('hub_todos', JSON.stringify(state.todos));
      renderTodos();
    }

    function deleteTodo(id) {
      state.todos = state.todos.filter(t => t.id !== id);
      localStorage.setItem('hub_todos', JSON.stringify(state.todos));
      renderTodos();
    }

    function renderTodos() {
      const list = document.getElementById('todo-list');
      list.innerHTML = '';
      
      const uncompletedCount = state.todos.filter(t => !t.done).length;
      document.getElementById('todo-badge').innerText = `${uncompletedCount} משימות לביצוע`;
      document.getElementById('count-todo-widget').innerText = uncompletedCount;

      if(state.todos.length === 0) {
        list.innerHTML = `<p class="text-xs text-slate-500 text-center py-6">אין משימות ברשימה. הגיע הזמן לנוח!</p>`;
        return;
      }

      state.todos.sort((a,b) => a.done - b.done).forEach(todo => {
        const item = document.createElement('div');
        item.className = `flex justify-between items-center p-3.5 rounded-xl border transition ${todo.done ? 'bg-slate-800/30 border-slate-800/80 opacity-60' : 'bg-slate-800/70 border-slate-700/40'}`;
        item.innerHTML = `
          <div class="flex items-center gap-3 flex-1 overflow-hidden" onclick="toggleTodo(${todo.id})">
            <div class="w-5 h-5 rounded-md border-2 flex items-center justify-center transition ${todo.done ? 'bg-blue-500 border-blue-500' : 'border-slate-500'}">
              ${todo.done ? '<svg xmlns="http://www.w3.org/2000/svg" class="h-3.5 w-3.5 text-white" viewBox="0 0 20 20" fill="currentColor"><path fill-rule="evenodd" d="M16.707 5.293a1 1 0 010 1.414l-8 8a1 1 0 01-1.414 0l-4-4a1 1 0 011.414-1.414L8 12.586l7.293-7.293a1 1 0 011.414 0z" clip-rule="evenodd" /></svg>' : ''}
            </div>
            <span class="text-sm truncate ${todo.done ? 'line-through text-slate-500' : 'text-slate-200'}">${todo.text}</span>
          </div>
          <button onclick="deleteTodo(${todo.id})" class="text-slate-500 hover:text-rose-400 p-1 mr-2">
            <svg xmlns="http://www.w3.org/2000/svg" class="h-4 w-4" fill="none" viewBox="0 0 24 24" stroke="currentColor"><path stroke-linecap="round" stroke-linejoin="round" stroke-width="2" d="M19 7l-.867 12.142A2 2 0 0116.138 21H7.862a2 2 0 01-1.995-1.858L5 7m5 4v6m4-6v6m1-10V4a1 1 0 00-1-1h-4a1 1 0 00-1 1v3M4 7h16" /></svg>
          </button>
        `;
        list.appendChild(item);
      });
    }

    // --- FINANCE FEATURE ---
    function addTransaction(type) {
      const amountEl = document.getElementById('finance-amount');
      const descEl = document.getElementById('finance-desc');
      const amount = parseFloat(amountEl.value);
      const desc = descEl.value.trim();

      if(!amount || !desc) {
        showToast("נא למלא סכום ותיאור מדויקים");
        return;
      }

      const tx = {
        id: Date.now(),
        amount: type === 'expense' ? -amount : amount,
        desc: desc,
        date: new Date().toLocaleDateString('he-IL', {month: 'numeric', day: 'numeric'})
      };

      state.transactions.unshift(tx);
      localStorage.setItem('hub_transactions', JSON.stringify(state.transactions));
      amountEl.value = '';
      descEl.value = '';
      renderFinance();
      showToast("התנועה נרשמה בהצלחה");
    }

    // --- DELETE TRANSACTION ---
    function deleteTransaction(id) {
      state.transactions = state.transactions.filter(t => t.id !== id);
      localStorage.setItem('hub_transactions', JSON.stringify(state.transactions));
      renderFinance();
    }

    function renderFinance() {
      const list = document.getElementById('finance-list');
      list.innerHTML = '';

      const total = state.transactions.reduce((acc, t) => acc + t.amount, 0);
      
      // Update balance UI strings
      const totalStr = `₪${total.toLocaleString('he-IL')}`;
      document.getElementById('finance-total-balance').innerText = totalStr;
      document.getElementById('balance-widget').innerText = totalStr;
      
      const widget = document.getElementById('balance-widget');
      if(total >= 0) {
        widget.className = "text-2xl font-black text-emerald-400";
      } else {
        widget.className = "text-2xl font-black text-rose-400";
      }

      if(state.transactions.length === 0) {
        list.innerHTML = `<p class="text-xs text-slate-500 text-center py-6">אין תנועות פיננסיות רשומות החודש.</p>`;
        return;
      }

      state.transactions.slice(0, 10).forEach(t => {
        const item = document.createElement('div');
        item.className = "flex justify-between items-center p-3 bg-slate-800/40 border border-slate-700/30 rounded-xl";
        const isIncome = t.amount > 0;
        item.innerHTML = `
          <div class="flex items-center gap-3 overflow-hidden">
            <span class="text-[10px] text-slate-500 font-mono">${t.date}</span>
            <span class="text-sm truncate font-medium text-slate-300">${t.desc}</span>
          </div>
          <div class="flex items-center gap-3">
            <span class="text-sm font-bold ${isIncome ? 'text-emerald-400' : 'text-rose-400'}">
              ${isIncome ? '+' : ''}₪${Math.abs(t.amount).toLocaleString('he-IL')}
            </span>
            <button onclick="deleteTransaction(${t.id})" class="text-slate-600 hover:text-rose-400">
              <svg xmlns="http://www.w3.org/2000/svg" class="h-3.5 w-3.5" fill="none" viewBox="0 0 24 24" stroke="currentColor"><path stroke-linecap="round" stroke-linejoin="round" stroke-width="2" d="M19 7l-.867 12.142A2 2 0 0116.138 21H7.862a2 2 0 01-1.995-1.858L5 7" /></svg>
            </button>
          </div>
        `;
        list.appendChild(item);
      });
    }

    // --- PRODUCTION GRADE PWA CONFIGURATION ---
    let deferredPrompt;
    const btnInstall = document.getElementById('btn-install');

    window.addEventListener('beforeinstallprompt', (e) => {
      e.preventDefault();
      deferredPrompt = e;
      btnInstall.classList.remove('hidden');
    });

    btnInstall.addEventListener('click', async () => {
      if (!deferredPrompt) return;
      deferredPrompt.prompt();
      const { outcome } = await deferredPrompt.userChoice;
      if (outcome === 'accepted') {
        showToast("תודה שהתקנת את SmartHub!");
      }
      deferredPrompt = null;
      btnInstall.classList.add('hidden');
    });

    window.addEventListener('appinstalled', (evt) => {
      showToast("🚀 האפליקציה הותקנה בהצלחה על המכשיר שלך!");
      btnInstall.classList.add('hidden');
    });

    if ('serviceWorker' in navigator) {

      `;
      const blob = new Blob([swCode], { type: 'application/javascript' });
      const swUrl = URL.createObjectURL(blob);
      navigator.serviceWorker.register(swUrl, { scope: './' })
        .then(reg => console.log('PWA ServiceWorker initialized successfully.'))
        .catch(err => console.log('SW setup note:', err));
